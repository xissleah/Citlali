cmake_minimum_required(VERSION 3.20)
project(Citlali VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
if(MINGW)
  add_link_options(-static-libgcc -static-libstdc++)
  get_filename_component(compiler_directory "${CMAKE_CXX_COMPILER}" DIRECTORY)
  find_file(CITLALI_WINPTHREAD libwinpthread-1.dll HINTS "${compiler_directory}" REQUIRED)
endif()
find_package(Threads REQUIRED)
option(CITLALI_BUILD_CPU_EXAMPLE "Build optional CPU-only GGUF example suite" OFF)
include(CTest)
add_library(citlali_core STATIC src/config.cpp src/host.cpp src/platform.cpp)
target_include_directories(citlali_core PUBLIC src sdk PRIVATE third_party)
target_link_libraries(citlali_core PUBLIC ${CMAKE_DL_LIBS} Threads::Threads)
add_executable(citlali src/main.cpp)
target_link_libraries(citlali PRIVATE citlali_core)
set_target_properties(citlali PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/dist/$<CONFIG>")
if(MINGW)
  add_custom_command(TARGET citlali POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CITLALI_WINPTHREAD}" "$<TARGET_FILE_DIR:citlali>/libwinpthread-1.dll")
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/dist/$<CONFIG>/mingw-winpthreads-LICENSE" INPUT "${CMAKE_SOURCE_DIR}/third_party/mingw-winpthreads-LICENSE")
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/dist/$<CONFIG>/plugins/.gitkeep" CONTENT "")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/dist/$<CONFIG>/LICENSE" INPUT "${CMAKE_SOURCE_DIR}/LICENSE" NEWLINE_STYLE UNIX)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/dist/$<CONFIG>/THIRD_PARTY.md" INPUT "${CMAKE_SOURCE_DIR}/third_party/THIRD_PARTY.md" NEWLINE_STYLE UNIX)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/dist/$<CONFIG>/README.md" INPUT "${CMAKE_SOURCE_DIR}/README.md" NEWLINE_STYLE UNIX)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/dist/$<CONFIG>/list.citlali" INPUT "${CMAKE_SOURCE_DIR}/list.citlali")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/dist/$<CONFIG>/tomlplusplus-LICENSE" INPUT "${CMAKE_SOURCE_DIR}/third_party/tomlplusplus-LICENSE")
if(BUILD_TESTING)
  add_subdirectory(tests/fixtures/lifecycle)
  add_executable(lifecycle_tests tests/lifecycle_tests.cpp)
  target_link_libraries(lifecycle_tests PRIVATE citlali_core)
  add_dependencies(lifecycle_tests lifecycle)
  add_test(NAME lifecycle_contracts COMMAND lifecycle_tests "${CMAKE_BINARY_DIR}/testing/$<CONFIG>")
  set_tests_properties(lifecycle_contracts PROPERTIES TIMEOUT 30)
  add_executable(abi_c tests/abi_c.c)
  target_include_directories(abi_c PRIVATE sdk)
  add_test(NAME abi_c COMMAND abi_c)
  add_executable(abi_cpp tests/abi_cpp.cpp)
  target_include_directories(abi_cpp PRIVATE sdk)
  add_test(NAME abi_cpp COMMAND abi_cpp)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_FOUND)
    add_test(NAME core_cli_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/core_cli.py" "$<TARGET_FILE:citlali>" "${CMAKE_BINARY_DIR}/dist/$<CONFIG>/list.citlali" "${CMAKE_BINARY_DIR}/testing/$<CONFIG>/plugins/lifecycle")
    set_tests_properties(core_cli_contracts PROPERTIES TIMEOUT 30)
  endif()
endif()
if(CITLALI_BUILD_CPU_EXAMPLE)
  add_subdirectory(plugins)
endif()
