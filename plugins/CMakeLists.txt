if(NOT WIN32 OR NOT MINGW OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "CPU example currently packages Windows x86_64 MinGW only")
endif()
set(CITLALI_LLAMA_SOURCE "" CACHE PATH "Matching llama.cpp source snapshot")
set(CITLALI_LLAMA_CPU_BUILD "" CACHE PATH "CPU-only backend built with tools/build_backend.py")
set(CITLALI_EXAMPLE_MODEL "" CACHE FILEPATH "Optional local GGUF model for tests")
set(CITLALI_EXAMPLE_TEST_MODELS "" CACHE STRING "Optional semicolon-separated GGUF paths for cross-model acceptance")
if(NOT EXISTS "${CITLALI_LLAMA_SOURCE}/include/llama.h" OR NOT EXISTS "${CITLALI_LLAMA_CPU_BUILD}/citlali-cpu.json")
  message(FATAL_ERROR "Provide llama source and a verified CPU build; see README.md")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/gguf-model/upstream.json" pin)
foreach(header include/llama.h ggml/include/ggml.h ggml/include/ggml-backend.h ggml/include/ggml-cpu.h ggml/include/ggml-opt.h)
  string(JSON expected GET "${pin}" headers_sha256 "${header}")
  file(SHA256 "${CITLALI_LLAMA_SOURCE}/${header}" actual)
  if(NOT expected STREQUAL actual)
    message(FATAL_ERROR "llama header snapshot mismatch: ${header}")
  endif()
endforeach()
file(READ "${CITLALI_LLAMA_CPU_BUILD}/citlali-cpu.json" backend)
foreach(header include/llama.h ggml/include/ggml.h ggml/include/ggml-backend.h ggml/include/ggml-cpu.h ggml/include/ggml-opt.h)
  string(JSON expected GET "${pin}" headers_sha256 "${header}")
  string(JSON built GET "${backend}" headers_sha256 "${header}")
  if(NOT expected STREQUAL built)
    message(FATAL_ERROR "CPU backend header provenance mismatch: ${header}")
  endif()
endforeach()
string(JSON cpu_only GET "${backend}" cpu_only)
if(NOT cpu_only)
  message(FATAL_ERROR "Expected CPU-only backend provenance")
endif()
set(backend_bin "${CITLALI_LLAMA_CPU_BUILD}/bin")
foreach(dll llama.dll ggml.dll ggml-base.dll ggml-cpu.dll)
  string(JSON expected GET "${backend}" dll_sha256 "${dll}")
  file(SHA256 "${backend_bin}/${dll}" actual)
  if(NOT expected STREQUAL actual)
    message(FATAL_ERROR "Backend changed: ${dll}")
  endif()
endforeach()
foreach(package gguf-model gguf-tokenizer gguf-runtime cli-basic)
  set(source "${CMAKE_CURRENT_SOURCE_DIR}/${package}")
  add_library(${package} SHARED "${source}/src/plugin.cpp")
  target_include_directories(${package} PRIVATE "${source}" "${PROJECT_SOURCE_DIR}/sdk" "${PROJECT_SOURCE_DIR}/third_party")
  target_link_libraries(${package} PRIVATE Threads::Threads)
  set(destination "${CMAKE_BINARY_DIR}/dist/$<CONFIG>/plugins/${package}")
  set_target_properties(${package} PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${destination}/bin")
  set(PLUGIN_ENTRY "bin/${package}${CMAKE_SHARED_LIBRARY_SUFFIX}")
  configure_file("${source}/info.toml" "${CMAKE_CURRENT_BINARY_DIR}/${package}-info.toml" @ONLY)
  file(GENERATE OUTPUT "${destination}/info.toml" INPUT "${CMAKE_CURRENT_BINARY_DIR}/${package}-info.toml")
  foreach(resource protocol.h protocol.md)
    file(GENERATE OUTPUT "${destination}/${resource}" INPUT "${source}/${resource}" NEWLINE_STYLE UNIX)
  endforeach()
  file(GLOB headers "${source}/include/*.h")
  foreach(header IN LISTS headers)
    get_filename_component(name "${header}" NAME)
    if(name STREQUAL "abi.h" OR name STREQUAL "text.h" OR name STREQUAL "frontend.h")
      set(header "${PROJECT_SOURCE_DIR}/sdk/${name}")
    endif()
    file(GENERATE OUTPUT "${destination}/include/${name}" INPUT "${header}" NEWLINE_STYLE UNIX)
  endforeach()
  foreach(license LICENSE llama.cpp-LICENSE)
    file(GENERATE OUTPUT "${destination}/${license}" INPUT "${source}/${license}" NEWLINE_STYLE UNIX)
  endforeach()
  file(GENERATE OUTPUT "${destination}/tomlplusplus-LICENSE" INPUT "${PROJECT_SOURCE_DIR}/third_party/tomlplusplus-LICENSE")
  file(GENERATE OUTPUT "${destination}/mingw-winpthreads-LICENSE" INPUT "${PROJECT_SOURCE_DIR}/third_party/mingw-winpthreads-LICENSE")
  add_custom_command(TARGET ${package} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CITLALI_WINPTHREAD}" "$<TARGET_FILE_DIR:${package}>/libwinpthread-1.dll")
endforeach()
target_sources(gguf-model PRIVATE "${PROJECT_SOURCE_DIR}/src/platform.cpp")
target_include_directories(gguf-model PRIVATE "${PROJECT_SOURCE_DIR}/src" "${CITLALI_LLAMA_SOURCE}/include" "${CITLALI_LLAMA_SOURCE}/ggml/include")
foreach(dll llama.dll ggml.dll ggml-base.dll ggml-cpu.dll libstdc++-6.dll libgcc_s_seh-1.dll)
  add_custom_command(TARGET gguf-model POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${backend_bin}/${dll}" "$<TARGET_FILE_DIR:gguf-model>/${dll}")
endforeach()
add_custom_command(TARGET gguf-model POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CITLALI_LLAMA_CPU_BUILD}/runtime-licenses" "$<TARGET_FILE_DIR:gguf-model>/../runtime-licenses")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/dist/$<CONFIG>/cpu-example/upstream.json" INPUT "${CMAKE_CURRENT_SOURCE_DIR}/gguf-model/upstream.json")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/dist/$<CONFIG>/cpu-example/backend.json" INPUT "${CITLALI_LLAMA_CPU_BUILD}/citlali-cpu.json")
if(BUILD_TESTING)
  foreach(package gguf-model gguf-tokenizer gguf-runtime cli-basic)
    add_library(example_headers_${package} OBJECT "${PROJECT_SOURCE_DIR}/tests/gguf-cpu/package_header.c" "${PROJECT_SOURCE_DIR}/tests/gguf-cpu/package_header.cpp")
    target_include_directories(example_headers_${package} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/${package}")
  endforeach()
  set(test_models ${CITLALI_EXAMPLE_TEST_MODELS})
  if(NOT test_models AND EXISTS "${CITLALI_EXAMPLE_MODEL}")
    set(test_models "${CITLALI_EXAMPLE_MODEL}")
  endif()
  if(Python3_FOUND)
    set(model_index 0)
    foreach(model IN LISTS test_models)
      if(NOT EXISTS "${model}")
        message(FATAL_ERROR "Acceptance model missing: ${model}")
      endif()
      math(EXPR model_index "${model_index}+1")
      set(test_name "gguf_cpu_acceptance_${model_index}")
      add_test(NAME ${test_name} COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/gguf-cpu/acceptance.py" "$<TARGET_FILE:citlali>" "${PROJECT_SOURCE_DIR}/README.md" "${model}" "${CMAKE_BINARY_DIR}/dist/$<CONFIG>/plugins")
      set_tests_properties(${test_name} PROPERTIES RUN_SERIAL TRUE TIMEOUT 300)
    endforeach()
  endif()
endif()
